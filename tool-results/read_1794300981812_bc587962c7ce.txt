    60→    60→function fmtIDR(v) {
    61→    61→    if (v >= 1e12) return 'Rp ' + (v / 1e12).toLocaleString('id-ID', { maximumFractionDigits: 2 }) + ' T';
    62→    62→    if (v >= 1e9) return 'Rp ' + (v / 1e9).toLocaleString('id-ID', { maximumFractionDigits: v < 1e10 ? 1 : 0 }) + ' M';
    63→    63→    if (v >= 1e6) return 'Rp ' + (v / 1e6).toLocaleString('id-ID', { maximumFractionDigits: v < 1e7 ? 1 : 0 }) + ' jt';
    64→    64→    if (v >= 1e3) return 'Rp ' + Math.round(v / 1e3) + ' rb';
    65→    65→    return 'Rp ' + NF.format(Math.round(v));
    66→    66→}
    67→    67→const fmtPrice = v => 'Rp ' + NF.format(Math.round(v));
    68→    68→const fmtInt = v => NF.format(Math.round(v));
    69→    69→function fmtVol(v) {
    70→    70→    if (v >= 1e9) return (v / 1e9).toLocaleString('id-ID', { maximumFractionDigits: 2 }) + ' M';
    71→    71→    if (v >= 1e6) return (v / 1e6).toLocaleString('id-ID', { maximumFractionDigits: 1 }) + ' jt';
    72→    72→    if (v >= 1e3) return (v / 1e3).toLocaleString('id-ID', { maximumFractionDigits: 0 }) + ' rb';
    73→    73→    return NF.format(Math.round(v));
    74→    74→}

    78→    78→function animateNumber(el, to, fmt, dur = 800) {
    79→    79→    if (!el) return;
    80→    80→    const from = Number(el.dataset.raw || 0);
    81→    81→    el.dataset.raw = to;
    82→    82→    const t0 = performance.now();
    83→    83→    (function frame(t) {
    84→    84→        const p = Math.min(1, (t - t0) / dur);
    85→    85→        const e = 1 - Math.pow(1 - p, 3);
    86→    86→        el.textContent = fmt(from + (to - from) * e);
    87→    87→        if (p < 1) requestAnimationFrame(frame);
    88→    88→    })(t0);
    89→    89→}

   112→   112→function niceCeil(v) {
   113→   113→    if (!(v > 0)) return 1;
   114→   114→    const p = Math.pow(10, Math.floor(Math.log10(v)));
   115→   115→    const f = v / p;
   116→   116→    const m = f <= 1 ? 1 : f <= 2 ? 2 : f <= 2.5 ? 2.5 : f <= 3 ? 3 : f <= 4 ? 4 : f <= 5 ? 5 : f <= 6 ? 6 : f <= 8 ? 8 : 10;
   117→   117→    return m * p;
   118→   118→}

   120→   120→function sample(arr, m) {
   121→   121→    if (!arr.length) return arr;
   122→   122→    if (arr.length <= m) return arr;
   123→   123→    const step = arr.length / m, out = [];
   124→   124→    for (let i = 0; i < m; i++) out.push(arr[Math.floor(i * step)]);
   125→   125→    return out;
   126→   126→}

   261→   261→function updateKPIs() {
   262→   262→    const d = state.data;
   263→   263→    if (!d.length) return;
   264→   264→    const rev = d.reduce((a, x) => a + x.total, 0);
   265→   265→    const vol = d.reduce((a, x) => a + x.vol, 0);
   266→   266→    const closeLast = d[d.length - 1].close;
   267→   267→    const s1 = d.reduce((a, x) => a + x.s1, 0);
   268→   268→    const s1Share = rev > 0 ? s1 / rev * 100 : 0;
   269→   269→    const p = state.prev;
   270→   270→    const dRev = p && p.total > 0 ? (rev - p.total) / p.total * 100 : null;
   271→   271→    const dOrd = p && p.vol > 0 ? (vol - p.vol) / p.vol * 100 : null;
   272→   272→    const dAov = p && p.closeLast > 0 ? (closeLast - p.closeLast) / p.closeLast * 100 : null;
   273→   273→    const pShare = p && p.total > 0 ? p.s1 / p.total * 100 : null;
   274→   274→    const dCnv = pShare != null ? s1Share - pShare : null;
   275→   275→
   276→   276→    state.kpi = { rev: Math.round(rev), orders: Math.round(vol), aov: Math.round(closeLast), conv: +s1Share.toFixed(1), dRev, dOrd, dAov, dCnv };
   277→   277→
   278→   278→    animateNumber($('#kRev'), rev, fmtIDR, 1000);
   279→   279→    animateNumber($('#kOrd'), vol, fmtInt, 1000);
   280→   280→    animateNumber($('#kAov'), closeLast, fmtPrice, 1000);
   281→   281→    animateNumber($('#kCnv'), s1Share, v => fmtPct(v), 1000);
   282→   282→
   283→   283→    setChip($('#dRev'), dRev, 'vs periode lalu');
   284→   284→    setChip($('#dOrd'), dOrd, 'vs periode lalu');
   285→   285→    setChip($('#dAov'), dAov, 'vs periode lalu');
   286→   286→    setChip($('#dCnv'), dCnv, 'poin vs periode lalu');
   287→   287→
   288→   288→    sparkline($('#spRev'), sample(d.map(x => x.total), 24), '#D14A1E');
   289→   289→    sparkline($('#spOrd'), sample(d.map(x => x.vol), 24), '#1C1712');
   290→   290→    sparkline($('#spAov'), sample(d.map(x => x.close), 24), '#1C1712');
   291→   291→    sparkline($('#spCnv'), sample(d.map(x => x.total > 0 ? x.s1 / x.total * 100 : 0), 24), '#1C1712');
   292→   292→}

   427→   427→function pollQuote() {

   442→   442→            setBadge(state.mode, state.asOf);

   506→   506→/* ============ MODE SIMULASI (fallback bila API offline) ============ */
   507→   507→function mulberry32(a) {
   508→   508→    return function() {
   509→   509→        a |= 0;
   510→   510→        a = a + 0x6D2B79F5 | 0;
   511→   511→        let t = Math.imul(a ^ a >>> 15, 1 | a);
   512→   512→        t = t + Math.imul(t ^ t >>> 7, 61 | t) ^ t;
   513→   513→        return ((t ^ t >>> 14) >>> 0) / 4294967296
   514→   514→    }
   515→   515→}

   517→   517→function startFallback() {
   518→   518→    state.mode = 'sim';
   519→   519→    const rng = mulberry32(Math.floor(Math.random() * 1e9));
   520→   520→    const n = state.days, now = Date.now(), out = [];
   521→   521→    for (let i = 0; i < n; i++) {
   522→   522→        const trend = 1 + 0.3 * i / Math.max(n - 1, 1);
   523→   523→        const s1 = 300e9 * trend * (0.8 + rng() * 0.4);
   524→   524→        const s2 = 180e9 * trend * (0.8 + rng() * 0.4);
   525→   525→        const close = 5200 + (rng() - 0.5) * 260;
   526→   526→        out.push({
   527→   527→            date: new Date(now - (n - 1 - i) * 864e5),
   528→   528→            dstr: new Date(now - (n - 1 - i) * 864e5).toISOString().slice(0, 10),
   529→   529→            s1, s2, total: s1 + s2, close, vol: Math.round((s1 + s2) / close)
   530→   530→        });
   531→   531→    }
   532→   532→    state.data = out;
   533→   533→    state.quote = { price: out[out.length - 1].close, changePct: 0, dayValue: out[out.length - 1].total, dayVolume: out[out.length - 1].vol, previousClose: out[out.length - 2]?.close || out[out.length - 1].close };
   534→   534→    state.prev = null;
   535→   535→    state.index = null;
   536→   536→    state.peers = [
   537→   537→        ['BBCA', 'Bank Central Asia'], ['BBRI', 'Bank Rakyat Indonesia'], ['BMRI', 'Bank Mandiri'],
   538→   538→        ['BBNI', 'Bank Negara Indonesia'], ['BRIS', 'Bank Syariah Indonesia'], ['ARTO', 'Bank Jago']
   539→   539→    ].map(([code, name]) => {
   540→   540→        const value = 200e9 + rng() * 300e9;
   541→   541→        return { code, name, value: Math.round(value), vol: Math.round(value / 5000), closeLast: 4000 + rng() * 2000, changePct: rng() * 10 - 3, dayChangePct: rng() * 3 - 1, share: 0 };
   542→   542→    }).sort((a, b) => b.value - a.value);
   543→   543→    const t = state.peers.reduce((a, p) => a + p.value, 0);
   544→   544→    state.peers.forEach(p => p.share = p.value / t * 100);
   545→   545→
   546→   546→    buildTicker();
   547→   547→    updateKPIs();
   548→   548→    renderChart();
   549→   549→    updateHero();
   550→   550→    renderPeers();
   551→   551→    renderSessions();
   552→   552→    seedFeed([]);
   553→   553→    setBadge('sim');
   554→   554→    toast('API tidak terjangkau — beralih ke MODE SIMULASI');
   555→   555→}
